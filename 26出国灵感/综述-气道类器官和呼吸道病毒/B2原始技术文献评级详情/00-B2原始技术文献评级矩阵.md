# 技术表现矩阵
```dataview
TABLE WITHOUT ID
file.link AS 文献,
choice(C1_评分 != null, C1_评分, C1_状态) AS C1,
choice(C2_评分 != null, C2_评分, C2_状态) AS C2,
choice(C3_评分 != null, C3_评分, C3_状态) AS C3,
choice(C5_评分 != null, C5_评分, C5_状态) AS C5,
choice(C6_评分 != null, C6_评分, C6_状态) AS C6,
choice(C7_评分 != null, C7_评分, C7_状态) AS C7,
choice(C8_评分 != null, C8_评分, C8_状态) AS C8,
choice(C9_评分 != null, C9_评分, C9_状态) AS C9
FROM "26出国灵感/综述-气道类器官和呼吸道病毒/B2原始技术文献评级详情"
WHERE file.name != this.file.name
SORT file.name ASC
```
