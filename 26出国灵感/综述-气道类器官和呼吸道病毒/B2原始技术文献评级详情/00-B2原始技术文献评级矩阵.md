# 技术表现矩阵
```dataview
TABLE WITHOUT ID
file.link AS 文献,
choice(C1.评分 != null, C1.评分, C1.状态) AS C1,
choice(C2.评分 != null, C2.评分, C2.状态) AS C2,
choice(C3.评分 != null, C3.评分, C3.状态) AS C3,
choice(C5.评分 != null, C5.评分, C5.状态) AS C5,
choice(C6.评分 != null, C6.评分, C6.状态) AS C6,
choice(C7.评分 != null, C7.评分, C7.状态) AS C7,
choice(C8.评分 != null, C8.评分, C8.状态) AS C8,
choice(C9.评分 != null, C9.评分, C9.状态) AS C9
FROM "26出国灵感/综述-气道类器官和呼吸道病毒/B2原始技术文献评级详情"
WHERE file.name != this.file.name
SORT file.name ASC
```
