```dataviewjs
const folderPath = '"03 行动管理"'
const noteTypes = ["学习计划", "实践计划", "项目计划"]
const data = dv.pages(folderPath)
	.filter(page => noteTypes.includes(page.file.frontmatter.noteType) && page.file.frontmatter.status == "已完成")
	.groupBy(page => page.file.mtime.toFormat("yyyy-MM-dd"))
	.map(entry => {
		return { date: entry.key, value: entry.rows.length }
	})   
    
const options = {
    data: data,
    days: 220,
    graphType: "calendar", // set this field value as 'month-track'
    cellStyleRules: [
        {
            color: "#fedcdc",
            min: 1,
            max: 3,
        },
        {
            color: "#f892a9",
            min: 3,
            max: 5,
        },
        {
            color: "#ec6a97",
            min: 5,
            max: 999,
        },
    ],
    cellStyle: {
        height: "14px",
        width: "14px"
    },
    showCellRuleIndicators: false,
    startOfWeek: 1,
}

renderContributionGraph(this.container, options)
```
