```dataviewjs
const folderPath = '"02 知识管理/01 输入"';

// 获取指定文件夹下所有页面的创建日期和对应的任务数量
const originalData = dv.pages(folderPath)
  .groupBy(page => page.file.ctime.toFormat("yyyy-MM-dd"))
  .map(entry => {
    return { date: entry.key, value: entry.rows.length };
  });

// 获取带有 #输入 标签的已完成任务
const inputTasksData = dv.pages()
  .flatMap(page => page.file.tasks)
  .where(t => t.text.includes("#输入") && t.completed)
  .map(task => {
    const completionDate = task.completion ? task.completion.toFormat("yyyy-MM-dd") : null;
    return completionDate ? { date: completionDate, value: 1 } : null;
  })
  .filter(Boolean);

// 合并数据源
const mergedData = [...originalData, ...inputTasksData];

// 去重并计算每天的任务总数
const mergedDataMap = mergedData.reduce((acc, item) => {
  acc[item.date] = (acc[item.date] || 0) + item.value;
  return acc;
}, {});

const finalData = Object.entries(mergedDataMap).map(([date, value]) => ({
  date: date,
  value: parseInt(value)
}));

const options = {
  data: finalData,
  days: 290,
  graphType: "calendar",
  cellStyleRules: [
    {
      color: "#00acff26",
      min: 1,
      max: 3,
    },
    {
      color: "#00acff8c",
      min: 3,
      max: 5,
    },
    {
      color: "#00acfff2",
      min: 5,
      max: 999,
    },
  ],
  cellStyle: {
    height: "14px",
    width: "11px"
  },
  showCellRuleIndicators: false,
  startOfWeek: 1,
};

renderContributionGraph(this.container, options);
```
