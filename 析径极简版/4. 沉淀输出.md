const folderPath = '"02 知识管理"'
const excludeFolder = '"02 知识管理/01 输入"'

const data = dv.pages(folderPath)
	.filter(page => page.file.frontmatter.status == "已完成" &&
	!page.file.path.startsWith(excludeFolder.slice(1, -1)))
	.groupBy(page => page.file.mtime.toFormat("yyyy-MM-dd"))
	.map(entry => {
		return { date: entry.key, value: entry.rows.length }
	})    
    
const options = {
    data: data,
    days: 290,
    graphType: "calendar", // set this field value as 'month-track'
    cellStyleRules: [
        {
            color: "#00acff26",
            min: 1,
            max: 3,
        },
        {
            color: "#00acff8c",
            min: 3,
            max: 5,
        },
        {
            color: "#00acfff2",
            min: 5,
            max: 999,
        },
    ],
    cellStyle: {
        height: "14px",
        width: "11px"
    },
    showCellRuleIndicators: false,
    startOfWeek: 1,
}

renderContributionGraph(this.container, options)
